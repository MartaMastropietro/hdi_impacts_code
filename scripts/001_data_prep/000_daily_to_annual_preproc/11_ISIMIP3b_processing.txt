            ww[5,2] <- ww[5,2] + 1 }  ## Close LOOP D2

           asd <- 0 } }  ## Close if (event) and LOOP D1
## MOST SEVERE EVENT (PAST) ###########################################################
         if ((sc == 1) & (ff <= 14)) { 
          for (k in 1:5) {  ## LOOP K1 - Five types of events
           if (ww[k,3] > wx[k,i,j]) wx[k,i,j] <- ww[k,3]
           asd <- 0 } }  ## Close if, LOOP K1
## STORE THE OUPUTS ###################################################################
         for (kk in 1:5) {  ## LOOP K1 - Five types of event
          for (pp in 1:4) {  ## LOOP P1 - Three (+1) outputs for each type of event (and year)
           CHW[kk,pp,i,j,yc] <- ww[kk,pp] }} }  ## Close  LOOPs P1-K1 and if on extended Hazard 1 (extd == 2)
#######################################################################################
### Reduced Hazard 1 - Two extremes ###################################################
#######################################################################################
        if (extd == 1)  {    
         ww[1:2,1:4] <- 0  ## Counters for output parameters
         ss[1:2,1:2] <- 0  ## Counters (sums) for core computations
## HAZ 1r.1 - NIGHT COLDWAVE (TNW in TTY[1,], TN10 in PTT[1,]) ########################
         jj <- 1
         for (ii in 1:(183-4)) {  ## LOOP D1 - Cold season (6 months)
          if (ii < jj) next   
          if ( (TTY[1,i,j,ii]   <= PTT[1,i,j,ii])   & (TTY[1,i,j,ii+1] <= PTT[1,i,j,ii+1]) & (TTY[1,i,j,ii+2] <= PTT[1,i,j,ii+2]) &
               (TTY[1,i,j,ii+3] <= PTT[1,i,j,ii+3]) & (TTY[1,i,j,ii+4] <= PTT[1,i,j,ii+4]) & max(TTY[1,i,j,ii:(ii+4)]) <= 0 ) {
# We included the additional condition: TN < 0°C for all the 5 days
           ww[1,1]   <- ww[1,1] + 1  ## Event counter
           ss[1,1:2] <- 0  ## Sum (record event?) and switch (unprec?)
           for (iii in ii:183) {  ## LOOP D2 - Inside the event
            jj <- iii + 1
            if ((TTY[1,i,j,iii] >  PTT[1,i,j,iii]) | (TTY[1,i,j,iii] > 0)) break  ## The event is over      
            ss[1,1] <- ss[1,1] + abs(TTY[1,i,j,iii] - PTT[1,i,j,iii])  ## Sum (event severity)
            if (ss[1,1] >= ww[1,3]) ww[1,3] <- ss[1,1]            ## If larger than the largest, this event becomes the largest
            if ((ff >= 15) & (ss[1,1] > wx[1,i,j])) ss[1,2] <- 1  ## Turn the switch on if the sum is larger than the largest event in the past (see below)
            ww[1,2] <- ww[1,2] + 1 }  ## Days within the event // Close LOOP D2 
           if (ss[1,2] == 1) ww[1,4] <- ww[1,4] + 1  ## We have an unprec event
           asd <- 0 } }  ## Close if (event) and LOOP D1
## HAZ 1r.2 - DAY HEATWAVE (TXS in TTY[2,..], TX90 in PTT[2,..]) #######################
         jj <- 1
         for (ii in 1:(183-4)) {  ## LOOP D1 - Hot season (6 months)
          if (ii < jj) next 
          if ( (TTY[2,i,j,ii]   >= PTT[2,i,j,ii])   & (TTY[2,i,j,ii+1] >= PTT[2,i,j,ii+1]) & (TTY[2,i,j,ii+2] >= PTT[2,i,j,ii+2]) &
               (TTY[2,i,j,ii+3] >= PTT[2,i,j,ii+3]) & (TTY[2,i,j,ii+4] >= PTT[2,i,j,ii+4]) & min(TTY[2,i,j,ii:(ii+4)]) >= 30 ) {
# We included the additional condition: TX > 30°C for all the 5 days
           ww[2,1]   <- ww[2,1] + 1
           ss[2,1:2] <- 0
           for (iii in ii:183) {  ## LOOP D2 - Inside the event
            jj <- iii + 1
            if ((TTY[2,i,j,iii] < PTT[2,i,j,iii]) | (TTY[2,i,j,iii] < 30)) break
            ss[2,1] <- ss[2,1] + abs(TTY[2,i,j,iii] - PTT[2,i,j,iii])
            if (ss[2,1] >= ww[2,3]) ww[2,3] <- ss[2,1]
            if ((ff >= 15) & (ss[2,1] > wx[2,i,j])) ss[2,2] <- 1  
            ww[2,2] <- ww[2,2] + 1 }  ## Close LOOP D2
           if (ss[2,2] == 1) ww[2,4] <- ww[2,4] + 1
           asd <- 0 } }  ## Close if (event) and LOOP D1
## MOST SEVERE EVENT (PAST) ###########################################################
         if ((sc == 1) & (ff <= 14)) { 
          for (k in 1:2) {  ## LOOP K1 - Two types of events
           if (ww[k,3] > wx[k,i,j]) wx[k,i,j] <- ww[k,3]
           asd <- 0 } }  ## Close if, LOOP K1
## STORE THE OUPUTS ###################################################################
         for (kk in 1:2) {  ## LOOP K1 - Two types of event
          for (pp in 1:4) {  ## LOOP P1 - Three (+1) outputs for each type of event (and year)
           CHW[kk,pp,i,j,yc] <- ww[kk,pp] }} } }  ## Close  LOOPs P1-K1 and if on reduced Haz 1 (extd == 1) and on Haz 1 (sw == 1)
#######################################################################################

#######################################################################################
### CORE HAZARD 2 #####################################################################
#######################################################################################
       if (sw == 2) {
        for (k in 1:nd) {  ## LOOP D1 - Days (Core loop)
         TM <- conv + 0.5*(TN[i,j,k] + TX[i,j,k])  ## TM (daily)
# NET EFFECTIVE TEMPERATURE
         NET <- 37 - 0.29*TM*(1 - 0.01*HU[i,j,k]) - (37 - TM)/(0.68 - 0.0014*HU[i,j,k] + 1/(1.76 + 1.4*(WN[i,j,k]^0.75)))
# VERY HOT AND COLD DAYS
         if (NET > 27) TC[1,i,j,yc] <- TC[1,i,j,yc] + 1  ## VERY HOT DAYS (VHD)
         if (NET <  1) TC[2,i,j,yc] <- TC[2,i,j,yc] + 1  ## VERY COLD DAYS (VCD)
         asd <- 0 } }  ## Close LOOP D1 and major if (sw == 2)
#######################################################################################

#######################################################################################
### CORE HAZARD 3 #####################################################################
#######################################################################################
       if (sw == 3) {
## BASE TEMPERATURES (UK MET-OFFICE)
        HTB <- 15.5  ## Base temperature for HDDs
        CTB <- 22    ## Base temperature for CDDs
## CORE LOOP ON DAYS
        for (k in 1:nd) {  ## LOOP D1 - Days (Current Year)
         TTN <- conv + TN[i,j,k]  ## TN in °C (daily value)
         TTX <- conv + TX[i,j,k]  ## TX in °C (daily value)
         TTM <- 0.5*(TTN + TTX)   ## TM in °C (daily value)
         if (TTN > TTX) next  ## Condition on bad data
## JAN-MAR and OCT-DEC (HDDs for NH, CDDs for SH)
         if ((k <= 91) | (k >= 274)) {
          if (j <= 180) {        ## Northern Hemisphere (Winter, Heating) 
           if  (TTX <= HTB)                DDD[1,i,j,yc] <- DDD[1,i,j,yc] + HTB - TTM
           if ((TTM <= HTB) & (TTX > HTB)) DDD[1,i,j,yc] <- DDD[1,i,j,yc] + 0.5*(HTB - TTN) - 0.25*(TTX - HTB)
           if ((TTN <  HTB) & (TTM > HTB)) DDD[1,i,j,yc] <- DDD[1,i,j,yc] + 0.25*(HTB - TTN) 
           if  (TTN >= HTB)                DDD[1,i,j,yc] <- DDD[1,i,j,yc]
          } else if (j > 180) {  ## Southern Hemisphere (Summer, Cooling) 
           if  (TTX <= CTB)                DDD[2,i,j,yc] <- DDD[2,i,j,yc]
           if ((TTM <= CTB) & (TTX > CTB)) DDD[2,i,j,yc] <- DDD[2,i,j,yc] + 0.25*(TTX - CTB)
           if ((TTN <  CTB) & (TTM > CTB)) DDD[2,i,j,yc] <- DDD[2,i,j,yc] + 0.5*(TTX - CTB) - 0.25*(CTB - TTN) 
           if  (TTN >= CTB)                DDD[2,i,j,yc] <- DDD[2,i,j,yc] + TTM - CTB 
           asd <- 0 }}
## APR-SEP (CDDs for NH, HDDs for SH)
         if ((k >= 91) & (k < 274)) {
          if (j <= 180) {        ## Northern Hemisphere (Summer, Cooling) 
           if  (TTX <= CTB)                DDD[2,i,j,yc] <- DDD[2,i,j,yc]
           if ((TTM <= CTB) & (TTX > CTB)) DDD[2,i,j,yc] <- DDD[2,i,j,yc] + 0.25*(TTX - CTB)
           if ((TTN <  CTB) & (TTM > CTB)) DDD[2,i,j,yc] <- DDD[2,i,j,yc] + 0.5*(TTX - CTB) - 0.25*(CTB - TTN) 
           if  (TTN >= CTB)                DDD[2,i,j,yc] <- DDD[2,i,j,yc] + TTM - CTB
          } else if (j > 180) {  ## Southern Hemisphere (Winter, Heating) 
           if  (TTX <= HTB)                DDD[1,i,j,yc] <- DDD[1,i,j,yc] + HTB - TTM
           if ((TTM <= HTB) & (TTX > HTB)) DDD[1,i,j,yc] <- DDD[1,i,j,yc] + 0.5*(HTB - TTN) - 0.25*(TTX - HTB)
           if ((TTN <  HTB) & (TTM > HTB)) DDD[1,i,j,yc] <- DDD[1,i,j,yc] + 0.25*(HTB - TTN) 
           if  (TTN >= HTB)                DDD[1,i,j,yc] <- DDD[1,i,j,yc] 
           asd <- 0 }} } ## Close LOOP D1
## JAN-DEC (EDDs for both NH and SH)
        DDD[3,i,j,yc] <- DDD[1,i,j,yc] + DDD[2,i,j,yc]}  ##  Close major if (sw == 3)
#######################################################################################

### NO HAZARD 4 HERE ##################################################################

#######################################################################################
### CORE HAZARD 5 #####################################################################
#######################################################################################
       if (sw == 5) {      
        ww[1:3] <- 0  ## Counters output parameters
        for (k in 1:nd) {  ## LOOP D1 - DAYS (col 9)
         if (is.finite(RR[i,j,k]) & (rain*RR[i,j,k] >= ATT[1,i,j,k])) ww[1] <- ww[1] + 1 
                                                                           ## EXTREME 1 - RR99 (days)
         if (k >= 5) {
          ss <- sum(rain*RR[i,j,k:(k-4)],na.rm=TRUE)                                      
          if (ss >= ww[2]) ww[2] <- ss                                     ## EXTREME 2 - RR5d (max)
          asd <- 0 } }  ## Close if and LOOP D1 (9)
# Most severe event from the past                 
        if ((sc == 1) & (ww[2] >= wx[i,j])) wx[i,j] <- ww[2]  
# Look for the unprecedented events in the future 
        if (ff >= 15) {
         kstop <- 5
         for (kk in 5:nd) {  ## LOOP D2 - DAYS (10)
          ks <- 0
          if (kk < kstop) next
          ks <- sum(rain*RR[i,j,kk:(kk-4)],na.rm=TRUE)
          if (ks > wx[i,j]) {                                              ## EXTREME 3 - RRu (ev)
           ww[3] <- ww[3] + 1                                      
           kstop <- kk + 5 } 
          asd <- 0 } }  ## Close LOOP D2 (10) and if (future; 9)
# Store the parameters into the outputs
        for (pp in 1:3) {  ## LOOP P1 - Three outputs (9)
         RRX[pp,i,j,yc] <- ww[pp] } }  ## Close LOOPs P1 (9) and if on sw == 5 (8)
#######################################################################################

#######################################################################################
### CORE HAZARD 6 (1/2) ###############################################################
#######################################################################################
       if (sw == 6) {
## HAZ 6.1 - NIGHT COLDWAVE ###########################################################
        jj <- 1
        for (ii in 1:(nd-4)) {  ## LOOP D1 - Days (365 days)
         my <- 12*(yc-1) + 1 + floor(ii/30.417) ## Position for the monthly outputs and the counters
         if (ii < jj) next 
         if (((conv + TN[i,j,ii])   <= ATT[2,i,j,ii])   & ((conv + TN[i,j,ii+1]) <= ATT[2,i,j,ii+1]) & 
             ((conv + TN[i,j,ii+2]) <= ATT[2,i,j,ii+2]) & ((conv + TN[i,j,ii+3]) <= ATT[2,i,j,ii+3]) &
             ((conv + TN[i,j,ii+4]) <= ATT[2,i,j,ii+4]) & (max(conv + TN[i,j,ii:(ii+4)]) <= 0)) {  ## Event detected
          for (iii in ii:nd) {  ## LOOP D2 - Inside the event
           jj <- iii + 1
           if (((conv + TN[i,j,iii]) > ATT[2,i,j,iii]) | ((conv + TN[i,j,iii]) > 0)) break  ## The event is over
           NNN[2,i,j,my] <- NNN[2,i,j,my] + 1
           asd <- 0 } } } ## Close LOOP D2, if (event), LOOP D1
## HAZ 6.2 - DAY HEATWAVE #############################################################
        jj <- 1
        for (ii in 1:(nd-4)) {  ## LOOP D1 - Days (365 days)
         my <- 12*(yc-1) + 1 + floor(ii/30.417)
         if (ii < jj) next
         if (((conv + TX[i,j,ii])   >= ATT[3,i,j,ii])   & ((conv + TX[i,j,ii+1]) >= ATT[3,i,j,ii+1]) & 
             ((conv + TX[i,j,ii+2]) >= ATT[3,i,j,ii+2]) & ((conv + TX[i,j,ii+3]) >= ATT[3,i,j,ii+3]) &
             ((conv + TX[i,j,ii+4]) >= ATT[3,i,j,ii+4]) & (min(conv + TX[i,j,ii:(ii+4)]) >= 30)) {  ## Event detected
          for (iii in ii:nd) {  ## LOOP D2 - Inside the event
           jj <- iii + 1
           if (((conv + TX[i,j,iii]) < ATT[3,i,j,iii]) | ((conv + TX[i,j,iii]) < 30)) break
           NNN[3,i,j,my] <- NNN[3,i,j,my] + 1 
           asd <- 0 } } } ## Close LOOP D2, if (event), LOOP D1
## HAZ 6.3 & 6.4 - EXTR RAINFALL DAY and EXTR WIND DAY ################################
        for (ii in 1:nd) {  ## LOOP D1 - Days (365 days)
         my <- 12*(yc-1) + 1 + floor(ii/30.417)
         if (rain*RR[i,j,ii] >= ATT[1,i,j,ii])  NNN[1,i,j,my] <- NNN[1,i,j,my] + 1
         if (WN[i,j,ii]      >= ATT[4,i,j,ii])  NNN[4,i,j,my] <- NNN[4,i,j,my] + 1
         asd <- 0 } } ## Close LOOP D1 and if on (sw == 6)
# Core calculations for Hazard 6 continue in a special sections after closing LOOPs 4-3
#######################################################################################

#######################################################################################
### CORE HAZARD 7 #####################################################################
#######################################################################################
       if (sw == 7) {
        for (ssn in 1:5) {  ## LOOP SS - Annual and Seasons (L9) // Not divided by Hemisphere 
         if (ssn == 2) {  ## "Winter" (months 12,1,2) // Seasonal sum for rain, average for temperature
          HCN[1,ssn,i,j,yc] <- rain*sum(RR[i,j,c(DSi[ssn]:DSf[ssn],DSi[6]:DSf[6])],na.rm=TRUE)    ## Kg/m2s into mm
          HCN[2,ssn,i,j,yc] <- conv + mean(TN[i,j,c(DSi[ssn]:DSf[ssn],DSi[6]:DSf[6])],na.rm=TRUE) ## °K into °C
          HCN[4,ssn,i,j,yc] <- conv + mean(TX[i,j,c(DSi[ssn]:DSf[ssn],DSi[6]:DSf[6])],na.rm=TRUE) ## °K into °C
          HCN[3,ssn,i,j,yc] <- 0.5*(HCN[2,ssn,i,j,yc] + HCN[4,ssn,i,j,yc])                        ## °C
         } else {          ## Annual or other seasons
          HCN[1,ssn,i,j,yc] <- rain*sum(RR[i,j,DSi[ssn]:DSf[ssn]],na.rm=TRUE)
          HCN[2,ssn,i,j,yc] <- conv + mean(TN[i,j,DSi[ssn]:DSf[ssn]],na.rm=TRUE)
          HCN[4,ssn,i,j,yc] <- conv + mean(TX[i,j,DSi[ssn]:DSf[ssn]],na.rm=TRUE)
          HCN[3,ssn,i,j,yc] <- 0.5*(HCN[2,ssn,i,j,yc] + HCN[4,ssn,i,j,yc]) } 
         asd <- 0 } } ## Close LOOP SS (9) and if on sw
#######################################################################################

#######################################################################################
### CORE HAZARD 8 (1/2) - MONTHLY VALUES ##############################################
#######################################################################################
       if (sw == 8) {  
        for (im in 1:12) { ## LOOP M1 - Months (9)  
         dii <- DMi[im]            ## First day of the month under investigation
         dff <- dii + MMi[im] - 1  ## Last day of the month under investigation
         mcm <- mc + im - 1        ## Position in the monthly outputs (month)
## Monthly RR (mm) and TN, TM, TX (°C)
         if (sc != 2) { 
          WCL[1,i,j,mcm] <- rain*sum(RR[i,j,dii:dff],na.rm=TRUE)     ## Monthly RR (mm)
          WCL[2,i,j,mcm] <- conv + mean(TN[i,j,dii:dff],na.rm=TRUE)  ## Monthly TN (°C)
          WCL[3,i,j,mcm] <- conv + mean(TX[i,j,dii:dff],na.rm=TRUE)  ## Monthly TX (°C)
          WCL[4,i,j,mcm] <- 0.5*(WCL[2,i,j,mcm] + WCL[3,i,j,mcm]) }  ## Monthly TM (°C) 
         if (sc == 2) {  ## Special case for hist-nat
          WCLn[1,i,j,mcm] <- rain*sum(RR[i,j,dii:dff],na.rm=TRUE)
          WCLn[2,i,j,mcm] <- conv + mean(TN[i,j,dii:dff],na.rm=TRUE)
          WCLn[3,i,j,mcm] <- conv + mean(TX[i,j,dii:dff],na.rm=TRUE)
          WCLn[4,i,j,mcm] <- 0.5*(WCLn[2,i,j,mcm] + WCLn[3,i,j,mcm]) } } ## Close if and LOOP M1 (9)
## Monthly PET (mm)
        if (sc != 2) WCL[5,i,j,mc:(mc+11)]  <- hargreaves(WCL[2,i,j,mc:(mc+11)],WCL[3,i,j,mc:(mc+11)],lat=LATy[i,j],na.rm=TRUE,verbose=FALSE)  
        if (sc == 2) WCLn[5,i,j,mc:(mc+11)] <- hargreaves(WCLn[2,i,j,mc:(mc+11)],WCLn[3,i,j,mc:(mc+11)],lat=LATy[i,j],na.rm=TRUE,verbose=FALSE) 
        asd <- 0 } ## Close if (sw)
#######################################################################################

#######################################################################################
### CORE HAZARD 9 #####################################################################
#######################################################################################
       if (sw == 9) {
# CALM DAYS (< 3 knots)
        WSP[1,i,j,yc] <- length(WN[i,j,1:nd][!is.na(WN[i,j,1:nd]) & WN[i,j,1:nd] < 1.5])
# OPTIMAL DAYS (FOR TURBINES; 3-10 m/s)
        WSP[2,i,j,yc] <- length(WN[i,j,1:nd][!is.na(WN[i,j,1:nd]) & WN[i,j,1:nd] >=  3]) -
                         length(WN[i,j,1:nd][!is.na(WN[i,j,1:nd]) & WN[i,j,1:nd] >= 18])     
# STORMY DAYS (> 10 m/s)
        WSP[3,i,j,yc] <- length(WN[i,j,1:nd][!is.na(WN[i,j,1:nd]) & WN[i,j,1:nd] >  10])
# CUMULATED VALUE (between 3-18 m/s)
        WSP[4,i,j,yc] <- sum(- 3 + WN[i,j,1:nd][WN[i,j,1:nd] >=  3],na.rm=TRUE) -
                         sum(- 3 + WN[i,j,1:nd][WN[i,j,1:nd] >= 18],na.rm=TRUE)
# ANNUAL AVERAGE (m/s)
        WSP[5,i,j,yc] <- mean(WN[i,j,1:nd],na.rm=TRUE) }  ## Close if on sw
#######################################################################################

#######################################################################################
### CORE HAZARD 10 ####################################################################
#######################################################################################
       if (sw == 10) {    
# Input/output variables for FWI.FBP package (re-initialized for every grid point)
        temp <- array(NA,c(nd))  ## TX
        rh   <- array(NA,c(nd))  ## HU
        ws   <- array(NA,c(nd))  ## WN (to avoid confusion, this "ws" is filled by WN data
        prec <- array(NA,c(nd))  ## RR
        lat  <- array(NA,c(nd))  ## LAT
        long <- array(NA,c(nd))  ## LON
# Assign monthly labels to days (We do for each year to split Haz 10 in multiple runs)
        for (k in 1:nd) {  ## LOOP D1 - DAYS (9)
#         if ((g + sc + ff + yy) == 4) {
         mon[k] <- 1 + floor(k/30.417)  ## Position for the monthly outputs and the counters
#          asd <- 0 }
# Convert variables into vectors for fwi function
         lat[k]  <- LATy[i,j]         ## LAT
         long[k] <- LONx[i,j]         ## LON
         temp[k] <- conv + TX[i,j,k]  ## TX (°K to °C)
         rh[k]   <- HU[i,j,k]         ## HU (%)
         ws[k]   <- wsp*WN[i,j,k]     ## WN (m/s to km/h)
         prec[k] <- rain*RR[i,j,k] }  ## RR (kgm2/s to mm/day) // ## Close LOOP D1 (9)
# Input dataframe for fwi function (vectors with 365 values)
        df <- data.frame(lat,long,mon,temp,rh,ws,prec)
# FWI with verbose R package (cffrds removed, so we used this package with a new pdated version)
        FW <- suppressWarnings(fwi(df,yda.fwi=NULL,init=c(ffmc_yda=85,dmc_yda=6,dc_yda=15),out="fwi",lat.adjust="TRUE"))
# Fire Danger values // FW has nd rows and 7 cols, COL 6 is FWI
        FDR[1,i,j,yc] <- length(FW[1:nd,6][!is.na(FW[1:nd,6]) & FW[1:nd,6] > 10])  ## MODERATE DANGER
        FDR[2,i,j,yc] <- length(FW[1:nd,6][!is.na(FW[1:nd,6]) & FW[1:nd,6] > 20])  ## HIGH DANGER
#        FDR[3,i,j,yc] <- length(FW[1:nd,6][!is.na(FW[1:nd,6]) & FW[1:nd,6] > 30])  ## EXTREME DANGER
        asd <- 0 } ## Close if (sw)
#######################################################################################

#######################################################################################
### CORE HAZARD 11 ####################################################################
#######################################################################################
       if (sw == 11) {
## HAZ 11.1 - GSL (Classic) ###########################################################
        switch <- 0  ## Are we inside the GSL? 
        ki <- 366  ## The GS does not start (hypothetic)
        km <- 184  ## First day of second semester (where GS can end)
        kf <- 365  ## The GS does not end (hypothetic)
        for (kk in 1:(nd-4)) {  ## LOOP D1 - DAYS (9)
         if (switch == 1) break  ## If the GSL already started, break at the next kk+ 
         if (min(TTY[2,i,j,kk:(kk+4)],na.rm=TRUE) >= 5) {  ## Condition to start GSL
          switch <- 1   ## GS started (and we assume it cannot end in the first semester)
          ki <- kk + 4  ## Start of the GSL (hypothetical non-start is replaced)
          if (ki >= (nd - 4)) break  ## If the start is from day 361, no end is feasible
          if (ki > km) km <- ki  ## We look for the end of GS in the second semester
          for (kkk in km:(nd-4)) {  ## LOOP E1 - Within the GS (11) . But only in second semester
           if (max(TTY[2,i,j,kkk:(kkk+4)],na.rm=TRUE) < 5) kf <- kkk + 3  ## End of the GSL
           if (max(TTY[2,i,j,kkk:(kkk+4)],na.rm=TRUE) < 5) break 
           asd <- 0 }  ## Close LOOP E1 (11)
          if (ki <= kf) GGG[1,i,j,yc] <- kf - ki + 1  ## Days in GS (Classic GSL)

       AKG[2,i,j,yy] <- 7                                      ## AI - Cold (7)
      } else if (XCL[3,i,j,yy] >= 10) {   ## Split (A to D)
       if (XCL[6,i,j,yy] > 0.7*XCL[1,i,j,yy]) {                ## KG - Special threshold on precipitation
        PTH <- 20*XCL[5,i,j,yy] + 280
       } else if (XCL[7,i,j,yy] > 0.7*XCL[1,i,j,yy]) {
        PTH <- 20*XCL[5,i,j,yy]
       } else {
        PTH <- 20*XCL[5,i,j,yy] + 140 }
       if (XCL[1,i,j,yy] <= PTH) {
        AKG[3,i,j,yy] <- 2                              ## KG - Arid (B;2) 
       } else {
        if (XCL[4,i,j,yy] <  -3)  AKG[3,i,j,yy] <- 4    ## KG - Continental (D;4)
        if (XCL[4,i,j,yy] >= -3)  AKG[3,i,j,yy] <- 3    ## KG - Temperate (C;3)
        if (XCL[4,i,j,yy] >  18)  AKG[3,i,j,yy] <- 1    ## KG - Tropical (A;1)
        asd <- 0 }} ## Close double if on KG
# Compute AI values
      AKG[1,i,j,yy] <- XCL[1,i,j,yy]/(XCL[2,i,j,yy] + 0.1)     ## AI - Values
# Compute AI classes
      if (AKG[2,i,j,yy] == 7) next                                
      if (AKG[1,i,j,yy] <  0.05)  AKG[2,i,j,yy] <- 1  ## AI - Hyperarid (1)
      if (AKG[1,i,j,yy] >= 0.05)  AKG[2,i,j,yy] <- 2  ## AI - Arid (2)
      if (AKG[1,i,j,yy] >= 0.2 )  AKG[2,i,j,yy] <- 3  ## AI - Semiarid (3)
      if (AKG[1,i,j,yy] >= 0.5 )  AKG[2,i,j,yy] <- 4  ## AI - Dry (4)
      if (AKG[1,i,j,yy] >= 0.65)  AKG[2,i,j,yy] <- 5  ## AI - Subhumid (5)
      if (AKG[1,i,j,yy] >  0.75)  AKG[2,i,j,yy] <- 6  ## AI - Humid (6) 
      asd <- 0 } }} }  ## Close LOOPs YY-L2-L1 (6-5-4), then if on sw
